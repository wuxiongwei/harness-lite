## 关键代码：settings.json 新 schema

```json
{
  "$schema": "https://json.schemastore.org/claude-code-settings.json",
  "permissions": {
    "allow": [
      "Read(./**)",
      "Glob(./**)",
      "Grep(./**)",
      "Bash(git status:*)",
      "Bash(git diff:*)",
      "Bash(git log:*)",
      "Bash(ls:*)",
      "Bash(pwd)",
      "Bash(find . *)"
    ],
    "deny": [
      "Bash(rm -rf *)",
      "Bash(rm -fr *)",
      "Bash(git push --force*)",
      "Bash(git push -f *)",
      "Bash(git reset --hard*)",
      "Bash(git clean -f*)",
      "Edit(**/.env)",
      "Edit(**/.env.*)",
      "Edit(**/credentials.*)",
      ...
    ]
  },
  "hooks": {
    "PreToolUse": [
      {
        "matcher": "Bash",
        "hooks": [
          {
            "type": "command",
            "if": "Bash(git commit*)",
            "command": "bash ${CLAUDE_PROJECT_DIR}/.claude/hooks/pre-commit-test.sh"
          },
          {
            "type": "command",
            "if": "Bash(git push*)",
            "command": "bash ${CLAUDE_PROJECT_DIR}/.claude/hooks/block-push.sh"
          }
        ]
      }
    ],
    "PostToolUse": [
      {
        "matcher": "Write",
        "hooks": [
          {
            "type": "command",
            "if": "Write(docs/versions/active/**/04-*.md)",
            "command": "bash ${CLAUDE_PROJECT_DIR}/.claude/hooks/distill-prompt.sh"
          }
        ]
      }
    ]
  }
}
```

## 关键代码：pre-commit-test.sh 退出码

```bash
# v1.0.4 → v1.0.5
- exit 1   # PreToolUse 不会因为 1 阻塞
+ exit 2   # 必须 2 才阻塞
```

阻塞信息打到 stderr，Claude Code 会展示给主 LLM。

## 关键代码：block-push.sh

```bash
#!/bin/bash
# Harness-Lite · block-push.sh
# AI 主动 git push 拦截器（exit 2 = PreToolUse 阻塞）
echo "🔴 Harness-Lite 门禁：禁止 AI 主动 push" >&2
echo "请用户 review 后手动执行 git push" >&2
exit 2
```

## 关键代码：distill-prompt.sh

```bash
#!/bin/bash
# Harness-Lite · distill-prompt.sh
# 检测 04-*.md 写入 → 提示沉淀知识到 team-knowledge/
input=$(cat)
fp=$(echo "$input" | jq -r '.tool_input.file_path // empty')
[ -z "$fp" ] && exit 0
echo "📚 Harness-Lite 提示：$fp 写入完成"
echo "   → 是否有踩坑/决策/接口值得沉淀？"
echo "   → pitfalls: team-knowledge/wiki/pitfalls/"
echo "   → decisions: team-knowledge/wiki/decisions/"
exit 0
```
